# ==============================================================================
# 1C:SRE-Suite - Ansible HA Module: Patroni + etcd + PostgreSQL + vip-manager
# ==============================================================================
# Модуль автоматизированного развертывания отказоустойчивого кластера СУБД под 1С.
# Данное решение полностью исключает промежуточное звено в виде HAProxy, 
# используя виртуальный IP (VIP) и утилиту vip-manager для прямой адресации.
#
# Все параметры и конфигурации оптимизированы под высокие нагрузки 1С:Предприятия.
# Лицензия: MIT
# ==============================================================================

# |-- hosts.ini                  # Справочник серверов и сетевых настроек
# |-- deploy-cluster.yml         # Главный сборочный плейбук

### 4. ШАБЛОН КОНФИГУРАЦИИ PATRONI (templates/patroni.yml.j2)
# Декларативный конфигурационный файл Patroni. 
# Настройки СУБД полностью адаптированы под требования 1С (блокировки, транзакции).

scope: cluster-1c
name: "{{ etcd_name }}"
namespace: /db/

log:
  format: '%(asctime)s %(levelname)s: %(message)s'
  level: INFO
  max_queue_size: 1000

restapi:
  listen: 0.0.0.0:8008
  connect_address: "{{ node_ip }}:8008"

etcd3:
  hosts:
    - 192.168.100.101:2379
    - 192.168.100.102:2379
    - 192.168.100.103:2379

bootstrap:
  dcs:
    loop_wait: 10
    retry_timeout: 10
    ttl: 30
    postgresql:
      use_pg_rewind: true
      use_slots: true
      parameters:
        hot_standby: "on"
        max_connections: 250
        max_locks_per_transaction: 256
        max_prepared_transactions: 0
        max_replication_slots: 10
        max_wal_senders: 10
        max_worker_processes: 8
        track_commit_timestamp: "off"
        wal_keep_size: 256MB
        wal_level: replica
        wal_log_hints: "on"
        # Специфичные настройки для планировщика 1С
        lc_messages: "ru_RU.UTF-8"
        shared_buffers: 4GB
        temp_buffers: 256MB

  initdb:
    - encoding: UTF8
    - data-checksums

  pg_hba:
    - host all all 0.0.0.0/0 md5
    - host replication patroni_replication 0.0.0.0/0 md5

postgresql:
  listen: 0.0.0.0:5432
  connect_address: "{{ node_ip }}:5432"
  bin_dir: "{{ pg_bin_dir }}"
  data_dir: "{{ pg_data_dir }}"
  authentication:
    superuser:
      username: patroni_superuser
      password: SuperuserSecretPassword
    replication:
      username: patroni_replication
      password: ReplicationSecretPassword
    rewind:
      username: patroni_rewind
      password: RewindSecretPassword

tags:
  clonefrom: true
  failover_priority: 1
  noloadbalance: false
  nosync: false


### 5. ШАБЛОН КОНФИГУРАЦИИ VIP-MANAGER (templates/vip-manager.yml.j2)

  - http://192.168.100.102:2379
  - http://192.168.100.103:2379

### 6. ШАБЛОН СЛУЖБЫ SYSTEMD PATRONI (templates/patroni.service.j2)
[Unit]
Description=Patroni Orchestrator for PostgreSQL HA
After=syslog.target network.target etcd.service

[Service]
Type=simple
User=postgres
Group=postgres
ExecStart=/usr/local/bin/patroni /etc/patroni/patroni.yml
ExecReload=/bin/kill -s HUP $MAINPID
KillMode=process
TimeoutSec=30
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target


### 7. СБОРОЧНЫЙ ПЛЕЙБУК (deploy-cluster.yml)
# Выполняет пошаговую установку пакетов, развертывание etcd, Patroni и vip-manager.
#
# Ключевые решения плейбука:
# - Включает net.ipv4.ip_nonlocal_bind для бинда виртуального IP на неактивных нодах.
# - Останавливает и отключает стандартный сервис СУБД, передавая управление Patroni.
# - Настраивает привилегированного пользователя patroni_rewind для отката СУБД.

- name: Deploy High-Availability PostgreSQL Cluster for 1C
  hosts: db_nodes
  become: true
  gather_facts: true

  tasks:
    - name: Ensure ip_nonlocal_bind is enabled (allows binding virtual IP)
      ansible.posix.sysctl:
        name: net.ipv4.ip_nonlocal_bind
        value: "1"
        sysctl_set: true
        state: present
        reload: true

    - name: Install dependencies (PostgreSQL Pro 1C and System Packages)
      ansible.builtin.package:
        name:
          - etcd
          - python3-pip
          - python3-psycopg2
          - iproute2
        state: present

    - name: Install Patroni via pip
      ansible.builtin.pip:
        name: patroni[etcd3]
        state: present

    - name: Ensure etcd configuration directory exists
      ansible.builtin.file:
        path: /etc/etcd
        state: directory
        owner: root
        group: root
        mode: "0755"

    - name: Deploy etcd configuration
      ansible.builtin.template:
        src: templates/etcd.conf.j2
        dest: /etc/etcd/etcd.conf
        owner: root
        group: root
        mode: "0644"
      notify: Restart etcd

    - name: Enable and start etcd service
      ansible.builtin.systemd:
        name: etcd
        enabled: true
        state: started

    - name: Stop and disable default PostgreSQL systemd service (managed by Patroni)
      ansible.builtin.systemd:
        name: postgrespro-1c-16
        enabled: false
        state: stopped
      failed_when: false

    - name: Ensure Patroni configuration directory exists
      ansible.builtin.file:
        path: /etc/patroni
        state: directory
        owner: postgres
        group: postgres
        mode: "0700"

    - name: Deploy Patroni configuration
      ansible.builtin.template:
        src: templates/patroni.yml.j2
        dest: /etc/patroni/patroni.yml
        owner: postgres
        group: postgres
        mode: "0600"

    - name: Deploy Patroni systemd service unit
      ansible.builtin.template:
        src: templates/patroni.service.j2
        dest: /etc/systemd/system/patroni.service
        owner: root
        group: root
        mode: "0644"

    - name: Enable and start Patroni service
      ansible.builtin.systemd:
        name: patroni
        daemon_reload: true
        enabled: true
        state: started

    - name: Download vip-manager package from official repository
      ansible.builtin.get_url:
        url: "https://github.com/cybertec-postgresql/vip-manager/releases/download/v5.0.0/vip-manager_5.0.0-1_amd64.deb"
        dest: "/tmp/vip-manager_5.0.0_amd64.deb"
        mode: "0644"

    - name: Install vip-manager package
      ansible.builtin.apt:
        deb: "/tmp/vip-manager_5.0.0_amd64.deb"
        state: present

    - name: Ensure vip-manager default configuration exists
      ansible.builtin.template:
        src: templates/vip-manager.yml.j2
        dest: /etc/default/vip-manager.yml
        owner: root
        group: root
        mode: "0600"
      notify: Restart vip-manager

    - name: Enable and start vip-manager service
      ansible.builtin.systemd:
        name: vip-manager
        enabled: true
        state: started

  handlers:
    - name: Restart etcd
      ansible.builtin.systemd:
        name: etcd
        state: restarted

    - name: Restart vip-manager
      ansible.builtin.systemd:
        name: vip-manager
        state: restarted
